class PayrollRecord(models.Model):
    class Status(models.TextChoices):
        DRAFT = "DRAFT", "Draft"
        LOCKED = "LOCKED", "Locked"
        PAID = "PAID", "Paid"

    employee = models.ForeignKey(
        "employees.Employee",
        on_delete=models.PROTECT,
        related_name="payroll_records",
    )

    year = models.PositiveIntegerField()
    month = models.PositiveSmallIntegerField()

    base_salary = models.DecimalField(
        max_digits=12,
        decimal_places=2,
    )

    lop_days = models.DecimalField(
        max_digits=5,
        decimal_places=2,
        default=0,
    )

    lop_deduction = models.DecimalField(
        max_digits=12,
        decimal_places=2,
        default=0,
    )

    net_salary = models.DecimalField(
        max_digits=12,
        decimal_places=2,
        default=0,
    )

    # Total earnings before deductions.
    gross_salary = models.DecimalField(
        max_digits=12,
        decimal_places=2,
        default=0,
    )

    # LOP deduction plus other salary deductions.
    total_deductions = models.DecimalField(
        max_digits=12,
        decimal_places=2,
        default=0,
    )

    # Snapshot of earnings and deductions used for this payroll.
    salary_components = models.JSONField(
        default=list,
        blank=True,
    )

    status = models.CharField(
        max_length=10,
        choices=Status.choices,
        default=Status.DRAFT,
    )

    generated_by = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.SET_NULL,
        null=True,
        blank=True,
    )

    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        constraints = [
            models.UniqueConstraint(
                fields=["employee", "year", "month"],
                name="unique_employee_payroll_per_month",
            ),
        ]
        ordering = ["-year", "-month"]

    def __str__(self):
        return (
            f"{self.employee.employee_code} - "
            f"{self.month}/{self.year} - {self.status}"
        )


class SalaryPayment(models.Model):
    # One simulated payment per payroll record.
    payroll_record = models.OneToOneField(
        "payroll.PayrollRecord",
        on_delete=models.PROTECT,
        related_name="salary_payment",
    )

    # Unique reference generated by the backend.
    reference = models.UUIDField(
        default=uuid.uuid4,
        unique=True,
        editable=False,
    )

    # Amount is calculated by the backend.
    amount = models.DecimalField(
        max_digits=12,
        decimal_places=2,
    )

    # This is a simulated payment, not a real bank transfer.
    status = models.CharField(
        max_length=20,
        default="SIMULATED_SUCCESS",
    )

    paid_at = models.DateTimeField(auto_now_add=True)

    processed_by = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.SET_NULL,
        null=True,
        blank=True,
        related_name="processed_salary_payments",
    )

    def __str__(self):
        return f"{self.reference} - {self.amount}"

CENT = Decimal("0.01")
ZERO = Decimal("0.00")


def money(value):
    """Convert a numeric value to a two-decimal Decimal."""
    return Decimal(str(value if value is not None else 0)).quantize(
        CENT,
        rounding=ROUND_HALF_UP,
    )


def send_payroll_notification(user, title, message):
    """Create a notification only when the user's preference permits it."""
    if user is None:
        return

    preferences, _ = NotificationPreference.objects.get_or_create(
        user=user
    )

    if not preferences.payroll_enabled:
        return

    # Avoid duplicate notifications for the same event.
    if Notification.objects.filter(
        recipient=user,
        notification_type="PAYROLL",
        title=title,
        message=message,
    ).exists():
        return

    Notification.objects.create(
        recipient=user,
        notification_type="PAYROLL",
        title=title,
        message=message,
    )


def weekdays_in_range(start_date, end_date):
    """Yield weekdays from Monday to Friday, inclusive."""
    current = start_date

    while current <= end_date:
        if current.weekday() < 5:
            yield current

        current += timedelta(days=1)


def get_overtime_summary(employee, start_date, end_date):
    """
    Summarize recorded overtime using assigned shift expected hours.

    This records hours only. It does not calculate overtime pay because
    the current implementation has no configured overtime payment rate.
    """
    assignments = list(
        ShiftAssignment.objects.filter(
            employee=employee,
            effective_start__lte=end_date,
        )
        .filter(
            models.Q(effective_end__isnull=True)
            | models.Q(effective_end__gte=start_date)
        )
        .select_related("shift")
        .order_by("-effective_start", "-created_at")
    )

    attendance_records = (
        Attendance.objects.filter(
            employee=employee,
            date__range=(start_date, end_date),
            check_out__isnull=False,
        )
        .order_by("date", "-check_in")
    )

    total_overtime = ZERO
    overtime_days = 0
    daily_details = []

    for attendance in attendance_records:
        assignment = next(
            (
                item
                for item in assignments
                if item.effective_start <= attendance.date
                and (
                    item.effective_end is None
                    or item.effective_end >= attendance.date
                )
            ),
            None,
        )

        # Do not invent expected hours for an employee without a shift.
        if assignment is None:
            continue

        expected_hours = Decimal(
            str(assignment.shift.expected_hours)
        )
        actual_hours = Decimal(
            str(attendance.working_hours or 0)
        )

        overtime_hours = max(
            actual_hours - expected_hours,
            ZERO,
        ).quantize(CENT, rounding=ROUND_HALF_UP)

        if overtime_hours <= 0:
            continue

        total_overtime += overtime_hours
        overtime_days += 1

        daily_details.append(
            {
                "date": attendance.date.isoformat(),
                "shift": assignment.shift.name,
                "expected_hours": str(expected_hours),
                "actual_hours": str(actual_hours),
                "overtime_hours": str(overtime_hours),
            }
        )

    return {
        "total_hours": total_overtime.quantize(
            CENT,
            rounding=ROUND_HALF_UP,
        ),
        "days": overtime_days,
        "details": daily_details,
    }


def calculate_payroll(employee, year, month, generated_by):
    """
    Calculate payroll for a completed month.

    Rules:
    - Current and future months cannot be generated.
    - Employees who joined after the month ended cannot have payroll
      generated for that month.
    - Weekends and configured holidays are excluded from LOP workdays.
    - Completed attendance and approved paid leave prevent LOP.
    - Unpaid leave and unexplained absences count as LOP.
    - Effective salary components are snapshotted in the payroll record.
    - Overtime hours are informational until an overtime pay rate exists.
    - Only draft payroll records can be recalculated.
    """
    today = timezone.localdate()

    if year < 1:
        raise ValueError("Year must be valid.")

    if not 1 <= month <= 12:
        raise ValueError("Month must be between 1 and 12.")

    if (year, month) >= (today.year, today.month):
        raise ValueError(
            "Payroll can only be generated for a completed month."
        )

    last_day = calendar.monthrange(year, month)[1]
    start_date = date(year, month, 1)
    end_date = date(year, month, last_day)

    joining_date = employee.joining_date

    if joining_date and joining_date > end_date:
        raise ValueError(
            "Payroll cannot be generated before the employee's joining date."
        )

    # Do not count any dates before the employee joined as absences.
    effective_start_date = max(
        start_date,
        joining_date or start_date,
    )

    holiday_dates = set(
        get_holiday_dates(
            employee,
            effective_start_date,
            end_date,
        )
    )

    work_dates = {
        day
        for day in weekdays_in_range(
            effective_start_date,
            end_date,
        )
        if day not in holiday_dates
    }

    attendance_dates = set(
        Attendance.objects.filter(
            employee=employee,
            date__range=(effective_start_date, end_date),
            check_out__isnull=False,
        ).values_list("date", flat=True)
    )

    approved_leaves = LeaveRequest.objects.filter(
        employee=employee,
        status=LeaveRequest.Status.APPROVED,
        start_date__lte=end_date,
        end_date__gte=effective_start_date,
    )

    paid_leave_dates = set()

    for leave in approved_leaves:
        # Only approved paid leave prevents LOP.
        if leave.leave_type == LeaveRequest.LeaveType.UNPAID:
            continue

        leave_start = max(
            leave.start_date,
            effective_start_date,
        )
        leave_end = min(leave.end_date, end_date)

        paid_leave_dates.update(
            day
            for day in weekdays_in_range(leave_start, leave_end)
            if day not in holiday_dates
        )

    # A completed attendance record takes precedence over paid leave.
    paid_leave_dates -= attendance_dates

    # Scheduled days without attendance or paid leave count as LOP.
    lop_dates = work_dates - attendance_dates - paid_leave_dates
    lop_days = Decimal(len(lop_dates))

    base_salary = money(employee.salary)

    if base_salary < ZERO:
        raise ValueError("Base salary cannot be negative.")

    total_workdays = len(work_dates)

    if total_workdays:
        per_day_salary = base_salary / Decimal(total_workdays)
        lop_deduction = money(per_day_salary * lop_days)
    else:
        lop_deduction = ZERO

    # Include components effective at any point during the payroll month.
    components = (
        EmployeeSalaryComponent.objects.filter(
            employee=employee,
            is_active=True,
            effective_from__lte=end_date,
        )
        .filter(
            models.Q(effective_to__isnull=True)
            | models.Q(effective_to__gte=effective_start_date)
        )
        .order_by("component_type", "name", "id")
    )

    component_snapshot = []
    total_earnings = ZERO
    other_deductions = ZERO

    for component in components:
        amount = money(component.amount)

        component_snapshot.append(
            {
                "id": component.pk,
                "name": component.name,
                "component_type": component.component_type,
                "amount": str(amount),
                "effective_from": component.effective_from.isoformat(),
                "effective_to": (
                    component.effective_to.isoformat()
                    if component.effective_to
                    else None
                ),
            }
        )

        if (
            component.component_type
            == EmployeeSalaryComponent.ComponentType.EARNING
        ):
            total_earnings += amount

        elif (
            component.component_type
            == EmployeeSalaryComponent.ComponentType.DEDUCTION
        ):
            other_deductions += amount

    total_earnings = money(total_earnings)
    other_deductions = money(other_deductions)

    overtime = get_overtime_summary(
        employee,
        effective_start_date,
        end_date,
    )

    if overtime["total_hours"] > ZERO:
        component_snapshot.append(
            {
                "name": "Overtime Hours",
                "component_type": "INFO",
                "amount": str(overtime["total_hours"]),
                "unit": "hours",
                "overtime_days": overtime["days"],
                "details": overtime["details"],
            }
        )

    gross_salary = money(base_salary + total_earnings)
    total_deductions = money(lop_deduction + other_deductions)
    net_salary = money(
        max(gross_salary - total_deductions, ZERO)
    )

    with transaction.atomic():
        record, created = PayrollRecord.objects.get_or_create(
            employee=employee,
            year=year,
            month=month,
            defaults={
                "base_salary": base_salary,
                "lop_days": lop_days,
                "lop_deduction": lop_deduction,
                "gross_salary": gross_salary,
                "total_deductions": total_deductions,
                "salary_components": component_snapshot,
                "net_salary": net_salary,
                "status": PayrollRecord.Status.DRAFT,
                "generated_by": generated_by,
            },
        )

        if not created:
            record = (
                PayrollRecord.objects.select_for_update()
                .get(pk=record.pk)
            )

            if record.status != PayrollRecord.Status.DRAFT:
                raise ValueError(
                    "Only draft payroll can be recalculated."
                )

            if SalaryPayment.objects.filter(
                payroll_record=record
            ).exists():
                raise ValueError(
                    "Payroll with a payment record cannot be recalculated."
                )

            record.base_salary = base_salary
            record.lop_days = lop_days
            record.lop_deduction = lop_deduction
            record.gross_salary = gross_salary
            record.total_deductions = total_deductions
            record.salary_components = component_snapshot
            record.net_salary = net_salary
            record.generated_by = generated_by

            record.save(
                update_fields=[
                    "base_salary",
                    "lop_days",
                    "lop_deduction",
                    "gross_salary",
                    "total_deductions",
                    "salary_components",
                    "net_salary",
                    "generated_by",
                    "updated_at",
                ]
            )

    return record


class GeneratePayrollView(APIView):
    permission_classes = [IsAdminOrHR]

    def post(self, request):
        try:
            employee_id = int(request.data.get("employee_id"))
            year = int(request.data.get("year"))
            month = int(request.data.get("month"))

            if employee_id <= 0:
                raise ValueError("Employee ID must be positive.")

        except (TypeError, ValueError):
            return Response(
                {
                    "detail": (
                        "Provide a valid employee_id, year and month."
                    )
                },
                status=status.HTTP_400_BAD_REQUEST,
            )

        try:
            employee = Employee.objects.select_related("user").get(
                pk=employee_id,
                is_active=True,
            )
        except Employee.DoesNotExist:
            return Response(
                {"detail": "Active employee not found."},
                status=status.HTTP_404_NOT_FOUND,
            )

        try:
            record = calculate_payroll(
                employee=employee,
                year=year,
                month=month,
                generated_by=request.user,
            )
        except ValueError as exc:
            return Response(
                {"detail": str(exc)},
                status=status.HTTP_400_BAD_REQUEST,
            )

        send_payroll_notification(
            user=employee.user,
            title="Payroll generated",
            message=(
                f"Your payroll for {calendar.month_name[record.month]} "
                f"{record.year} has been generated. "
                f"Net salary: Rs. {record.net_salary:,.2f}."
            ),
        )

        return Response(
            PayrollRecordSerializer(record).data,
            status=status.HTTP_201_CREATED,
        )


class MyPayrollHistoryView(generics.ListAPIView):
    serializer_class = PayrollRecordSerializer
    permission_classes = [IsAuthenticated]

    def get_queryset(self):
        employee = get_object_or_404(
            Employee,
            user=self.request.user,
        )

        queryset = (
            PayrollRecord.objects.filter(employee=employee)
            .select_related(
                "employee",
                "employee__user",
                "generated_by",
            )
            .order_by("-year", "-month")
        )

        year = self.request.query_params.get("year")
        month = self.request.query_params.get("month")

        if year:
            if not year.isdigit():
                return queryset.none()
            queryset = queryset.filter(year=int(year))

        if month:
            if not month.isdigit() or not 1 <= int(month) <= 12:
                return queryset.none()
            queryset = queryset.filter(month=int(month))

        return queryset

class LockPayrollView(APIView):
    permission_classes = [IsAdminOrHR]

    def post(self, request, pk):
        with transaction.atomic():
            record = get_object_or_404(
                PayrollRecord.objects.select_for_update().select_related(
                    "employee",
                    "employee__user",
                ),
                pk=pk,
            )

            if record.status != PayrollRecord.Status.DRAFT:
                return Response(
                    {"detail": "Only draft payroll can be locked."},
                    status=status.HTTP_400_BAD_REQUEST,
                )

            if SalaryPayment.objects.filter(
                payroll_record=record
            ).exists():
                return Response(
                    {
                        "detail": (
                            "Payroll already has a payment record "
                            "and cannot be locked again."
                        )
                    },
                    status=status.HTTP_400_BAD_REQUEST,
                )

            record.status = PayrollRecord.Status.LOCKED
            record.save(update_fields=["status", "updated_at"])

        send_payroll_notification(
            user=record.employee.user,
            title="Payroll finalized",
            message=(
                f"Your payroll for {calendar.month_name[record.month]} "
                f"{record.year} has been locked and finalized. "
                f"Net salary: Rs. {record.net_salary:,.2f}."
            ),
        )

        return Response(PayrollRecordSerializer(record).data)

                pk=pk,
            )

            if record.status != PayrollRecord.Status.LOCKED:
                return Response(
                    {"detail": "Only locked payroll can be paid."},
                    status=status.HTTP_400_BAD_REQUEST,
                )

            if SalaryPayment.objects.filter(
                payroll_record=record
            ).exists():
                return Response(
                    {
                        "detail": (
                            "A payment record already exists for this payroll."
                        )
                    },
                    status=status.HTTP_400_BAD_REQUEST,
                )

            payment = SalaryPayment.objects.create(
                payroll_record=record,
                amount=record.net_salary,
                processed_by=request.user,
                status="SIMULATED_SUCCESS",
            )

            record.status = PayrollRecord.Status.PAID
            record.save(update_fields=["status", "updated_at"])

        send_payroll_notification(
            user=record.employee.user,
            title="Salary payment simulated",
            message=(
                f"The simulated salary payment for "
                f"{calendar.month_name[record.month]} {record.year} "
                f"has been recorded. Amount: Rs. "
                f"{record.net_salary:,.2f}. No real bank transfer was made."
            ),
        )

        return Response(
            {
                "message": (
                    "Simulated salary credit recorded. "
                    "No real bank transfer was made."
                ),
                "payment": SalaryPaymentSerializer(payment).data,
                "payroll": PayrollRecordSerializer(record).data,
            },
            status=status.HTTP_201_CREATED,
        )

from .models import Holiday, OptionalHolidaySelection


def get_holiday_dates(employee, start_date, end_date):
    """
    Return the dates of holidays applicable to an employee.

    Rules:
    1. Company holidays apply to every employee.
    2. Regional holidays apply when the employee's region matches.
    3. Optional holidays apply only when the employee selected them.
    4. Inactive holidays are ignored.
    """

    if not employee or start_date > end_date:
        return set()

    # Fetch active holidays within the requested date range.
    holidays = Holiday.objects.filter(
        is_active=True,
        date__range=(start_date, end_date),
    )

    # Company-wide holidays apply to everyone.
    applicable_dates = set(
        holidays.filter(
            holiday_type=Holiday.HolidayType.COMPANY,
        ).values_list("date", flat=True)
    )

    # Regional holidays apply only to employees in that region.
    employee_region = (employee.region or "").strip()

    if employee_region:
        regional_dates = holidays.filter(
            holiday_type=Holiday.HolidayType.REGIONAL,
            region__iexact=employee_region,
        ).values_list("date", flat=True)

        applicable_dates.update(regional_dates)

    # Optional holidays count only when this employee selected them.
    selected_optional_dates = OptionalHolidaySelection.objects.filter(
        employee=employee,
        holiday__is_active=True,
        holiday__holiday_type=Holiday.HolidayType.OPTIONAL,
        holiday__date__range=(start_date, end_date),
    ).values_list("holiday__date", flat=True)

    applicable_dates.update(selected_optional_dates)

    return applicable_dates

function getErrorMessage(data, fallback) {
  if (typeof data?.detail === "string") return data.detail;

  if (data && typeof data === "object") {
    const messages = Object.entries(data).map(([field, value]) => {
      const message = Array.isArray(value)
        ? value.map(String).join(" ")
        : typeof value === "string"
          ? value
          : JSON.stringify(value);

      return `${field}: ${message}`;
    });

    if (messages.length) return messages.join(" ");
  }

  return fallback;
}

async function apiRequest(path, accessToken, options = {}) {
  const response = await fetch(`${API_BASE}${path}`, {
    ...options,
    headers: {
      ...(options.body ? { "Content-Type": "application/json" } : {}),
      ...options.headers,
      ...(accessToken ? { Authorization: `Bearer ${accessToken}` } : {}),
    },
  });

  if (!response.ok) {
    const data = await response.json().catch(() => ({}));
    throw new Error(
      getErrorMessage(data, `Request failed (${response.status})`),
    );
  }

  if (response.status === 204) return null;

  const contentType = response.headers.get("content-type") || "";
  if (contentType.includes("application/json")) {
    return response.json();
  }

  return response;
}

function getMonthName(month) {
  const value = Number(month);
  if (!Number.isInteger(value) || value < 1 || value > 12) {
    return "Unknown month";
  }

  return new Date(2000, value - 1, 1).toLocaleString("en-IN", {
    month: "long",
  });
}

function PayrollStatus({ status }) {
  const styles = {
    DRAFT: "bg-amber-100 text-amber-800",
    LOCKED: "bg-blue-100 text-blue-800",
    PAID: "bg-green-100 text-green-800",
  };

  return (
    <span
      className={`inline-flex rounded-full px-3 py-1 text-xs font-bold ${
        styles[status] || "bg-slate-100 text-slate-700"
      }`}
    >
      {status || "UNKNOWN"}
    </span>
  );
}

function SummaryCard({ label, value, icon: Icon }) {
  return (
    <div className="rounded-2xl border border-slate-200 bg-white p-5 shadow-sm">
      <div className="flex items-center justify-between gap-3">
        <p className="text-sm text-slate-500">{label}</p>
        <Icon size={20} className="text-blue-700" />
      </div>
      <p className="mt-3 text-3xl font-bold text-slate-900">{value}</p>
    </div>
  );
}

export default function PayrollManagement({
  user,
  role,
  accessToken,
  onBack,
  onLogout,
}) {
  const currentRole = (
    role ||
    user?.role ||
    user?.profile?.role ||
    user?.hrms_profile?.role ||
    (user?.is_superuser ? "ADMIN" : "EMPLOYEE")
  )
    .toString()
    .toUpperCase();

  const isAdminHR = ["ADMIN", "HR", "SUPERUSER"].includes(currentRole);

  const today = new Date();
  const previousMonth = new Date(
    today.getFullYear(),
    today.getMonth() - 1,
    1,
  );

  const [year, setYear] = useState(String(previousMonth.getFullYear()));
  const [month, setMonth] = useState(String(previousMonth.getMonth() + 1));
  const [employeeId, setEmployeeId] = useState("");
  const [employees, setEmployees] = useState([]);
  const [records, setRecords] = useState([]);
  const [loading, setLoading] = useState(true);
  const [generating, setGenerating] = useState(false);
  const [workingId, setWorkingId] = useState(null);
  const [downloadingId, setDownloadingId] = useState(null);
  const [error, setError] = useState("");
  const [success, setSuccess] = useState("");
  const [search, setSearch] = useState("");
  const [statusFilter, setStatusFilter] = useState("ALL");
  const [periodFilter, setPeriodFilter] = useState("ALL");

  const loadRecords = useCallback(async () => {
    setLoading(true);
    setError("");

    try {
      const path = isAdminHR ? "/payroll/" : "/payroll/my/";
      const data = await apiRequest(path, accessToken);
      setRecords(getRows(data));
    } catch (err) {
      setError(err.message || "Unable to load payroll records.");
    } finally {
      setLoading(false);
    }
  }, [accessToken, isAdminHR]);

  const loadEmployees = useCallback(async () => {
    if (!isAdminHR) return;

    try {
      const data = await apiRequest("/employees/", accessToken);
      const activeEmployees = getRows(data).filter(
        (employee) => employee.is_active !== false,
      );

      setEmployees(activeEmployees);

      setEmployeeId((current) => {
        if (
          current &&
          activeEmployees.some(
            (employee) => String(employee.id) === String(current),
          )
        ) {
          return current;
        }

        return activeEmployees.length
          ? String(activeEmployees[0].id)
          : "";
      });
    } catch (err) {
      setError(
        `Could not load employees: ${err.message || "Request failed"}`,
      );
    }
  }, [accessToken, isAdminHR]);

  useEffect(() => {
    loadRecords();
    loadEmployees();
  }, [loadRecords, loadEmployees]);

  async function refreshAll() {
    setSuccess("");
    await Promise.all([loadRecords(), loadEmployees()]);
  }

  async function generatePayroll(event) {
    event.preventDefault();
    setError("");
    setSuccess("");

    if (!accessToken) {
      setError("Your session is missing. Please log in again.");
      return;
    }

    if (!employeeId) {
      setError("Select an employee before generating payroll.");
      return;
    }

    const selectedYear = Number(year);
    const selectedMonth = Number(month);

    if (
      !Number.isInteger(selectedYear) ||
      selectedYear < 1 ||
      !Number.isInteger(selectedMonth) ||
      selectedMonth < 1 ||
      selectedMonth > 12
    ) {
      setError("Enter a valid payroll year and month.");
      return;
    }

    if (
      selectedYear > today.getFullYear() ||
      (selectedYear === today.getFullYear() &&
        selectedMonth >= today.getMonth() + 1)
    ) {
      setError("Select a completed month for payroll generation.");
      return;
    }

    setGenerating(true);

    try {
      await apiRequest("/payroll/generate/", accessToken, {
        method: "POST",
        body: JSON.stringify({
          employee_id: Number(employeeId),
          year: selectedYear,
          month: selectedMonth,
        }),
      });

      setSuccess(
        `Payroll generated or recalculated for ${getMonthName(selectedMonth)} ${selectedYear}.`,
      );

      await loadRecords();
    } catch (err) {
      setError(err.message || "Unable to generate payroll.");
    } finally {
      setGenerating(false);
    }
  }

  async function changePayrollStatus(record, action) {
    const actions = {
      lock: {
        prompt:
          "Lock this draft payroll? A locked record cannot be recalculated.",
        path: `/payroll/${record.id}/lock/`,
        successMessage: "Payroll locked successfully.",
      },
      pay: {
        prompt:
          "Record a simulated salary payment? This will not transfer real money.",
        path: `/payroll/${record.id}/pay/`,
        successMessage:
          "Simulated salary payment recorded. No real bank transfer occurred.",
      },
    };

    const details = actions[action];
    if (!details || workingId !== null) return;

    if (!window.confirm(details.prompt)) return;

    setWorkingId(record.id);
    setError("");
    setSuccess("");

    try {
      await apiRequest(details.path, accessToken, {
        method: "POST",
        body: JSON.stringify({}),
      });

      setSuccess(details.successMessage);
      await loadRecords();
    } catch (err) {
      setError(err.message || "Unable to update payroll status.");
    } finally {
      setWorkingId(null);
    }
  }

  async function downloadSalarySlip(record) {
    setError("");
    setSuccess("");
    setDownloadingId(record.id);

    try {
      const response = await fetch(
        `${API_BASE}/payroll/${record.id}/salary-slip/`,
        {
          headers: accessToken
            ? { Authorization: `Bearer ${accessToken}` }
            : {},
        },
      );

      if (!response.ok) {
        const data = await response.json().catch(() => ({}));
        throw new Error(
          getErrorMessage(data, `Download failed (${response.status})`),
        );
      }

      const blob = await response.blob();
      const url = URL.createObjectURL(blob);
      const link = document.createElement("a");

      link.href = url;
      link.download =
        `salary-slip-${record.employee_code || record.id}-` +
        `${record.year}-${String(record.month).padStart(2, "0")}.pdf`;

      document.body.appendChild(link);
      link.click();
      link.remove();

      // Revoke after the browser has had an opportunity to start the download.
      window.setTimeout(() => URL.revokeObjectURL(url), 1000);

      setSuccess("Salary slip download started.");
    } catch (err) {
      setError(err.message || "Unable to download the salary slip.");
    } finally {
      setDownloadingId(null);
    }
  }

  const visibleRecords = useMemo(() => {
    const query = search.trim().toLowerCase();

    return records.filter((record) => {
      const matchesSearch =
        !query ||
        String(record.employee_name || "")
          .toLowerCase()
          .includes(query) ||
        String(record.employee_code || "")
          .toLowerCase()
          .includes(query);

      const matchesStatus =
        statusFilter === "ALL" || record.status === statusFilter;

      const matchesPeriod =
        periodFilter === "ALL" ||
        `${record.year}-${String(record.month).padStart(2, "0")}` ===
          periodFilter;

      return matchesSearch && matchesStatus && matchesPeriod;
    });
  }, [records, search, statusFilter, periodFilter]);

  const periods = useMemo(() => {
    const unique = new Set(
      records.map(
        (record) =>
          `${record.year}-${String(record.month).padStart(2, "0")}`,
      ),
    );

    return [...unique].sort((a, b) => b.localeCompare(a));
  }, [records]);

  const summary = useMemo(
    () => ({
      count: visibleRecords.length,
      draft: visibleRecords.filter((r) => r.status === "DRAFT").length,
      locked: visibleRecords.filter((r) => r.status === "LOCKED").length,
      paid: visibleRecords.filter((r) => r.status === "PAID").length,
      net: visibleRecords.reduce(
        (sum, record) => sum + Number(record.net_salary || 0),
        0,
      ),
    }),
    [visibleRecords],
  );

  return (
    <main className="min-h-screen bg-slate-100 text-slate-900">
      <header className="border-b border-slate-200 bg-white">
        <div className="mx-auto flex max-w-7xl flex-wrap items-center justify-between gap-3 px-5 py-4">
          <div className="flex items-center gap-3">
            <button
              type="button"
              onClick={onBack}
              className="rounded-lg border border-slate-300 p-2 hover:bg-slate-50"
              aria-label="Back to dashboard"
            >
              <ArrowLeft size={20} />
            </button>

            <div>
              <h1 className="text-xl font-bold">Payroll Management</h1>
              <p className="text-sm text-slate-500">
                Salary calculation, payroll approvals, and salary slips
              </p>
            </div>
          </div>

          <button
            type="button"
            onClick={onLogout}
            className="rounded-lg border border-slate-300 px-3 py-2 text-sm font-semibold hover:bg-slate-50"
          >
            Log out
          </button>
        </div>
      </header>

      <div className="mx-auto max-w-7xl space-y-6 px-5 py-8">
        {error && (
          <div
            role="alert"
            className="flex items-start gap-2 rounded-xl border border-red-200 bg-red-50 p-4 text-sm text-red-800"
          >
            <AlertCircle size={18} className="mt-0.5 shrink-0" />
            <span>{error}</span>
          </div>
        )}

        {success && (
          <div
            role="status"
            className="flex items-start gap-2 rounded-xl border border-green-200 bg-green-50 p-4 text-sm text-green-800"
          >
            <CheckCircle2 size={18} className="mt-0.5 shrink-0" />
            <span>{success}</span>
          </div>
        )}

        {isAdminHR && (
          <section className="rounded-2xl border border-slate-200 bg-white p-6 shadow-sm">
            <div className="mb-5 flex items-center gap-3">
              <div className="rounded-xl bg-blue-50 p-3 text-blue-700">
                <Wallet size={23} />
              </div>
              <div>
                <h2 className="text-lg font-bold">Generate payroll</h2>
                <p className="text-sm text-slate-500">
                  Generate or recalculate payroll for a completed month.
                </p>
              </div>
            </div>

            <form
              onSubmit={generatePayroll}
              className="grid gap-4 sm:grid-cols-2 lg:grid-cols-4"
            >
              <div className="sm:col-span-2">
                <label
                  htmlFor="payroll_employee"
                  className="mb-1.5 block text-sm font-semibold text-slate-700"
                >
                  Employee
                </label>
                <select
                  id="payroll_employee"
                  value={employeeId}
                  onChange={(event) => setEmployeeId(event.target.value)}
                  required
                  className="w-full rounded-lg border border-slate-300 bg-white px-3 py-2.5"
                >
                  <option value="">Select employee</option>
                  {employees.map((employee) => (
                    <option key={employee.id} value={employee.id}>
                      {employee.employee_code} —{" "}
                      {employee.full_name ||
                        employee.employee_name ||
                        employee.username ||
                        employee.user?.username ||
                        `Employee ${employee.id}`}
                    </option>
                  ))}
                </select>

                {employees.length === 0 && (
                  <p className="mt-1 text-xs text-amber-700">
                    No employees found. Check your employee API and active
                    employee records.
                  </p>
                )}
              </div>

              <div>
                <label
                  htmlFor="payroll_year"
                  className="mb-1.5 block text-sm font-semibold text-slate-700"
                >
                  Year
                </label>
                <input
                  id="payroll_year"
                  type="number"
                  min="2000"
                  max={today.getFullYear()}
                  value={year}
                  onChange={(event) => setYear(event.target.value)}
                  required
                  className="w-full rounded-lg border border-slate-300 px-3 py-2.5"
                />
              </div>

              <div>
                <label
                  htmlFor="payroll_month"
                  className="mb-1.5 block text-sm font-semibold text-slate-700"
                >
                  Month
                </label>
                <select
                  id="payroll_month"
                  value={month}
                  onChange={(event) => setMonth(event.target.value)}
                  required
                  className="w-full rounded-lg border border-slate-300 bg-white px-3 py-2.5"
                >
                  {Array.from({ length: 12 }, (_, index) => index + 1).map(
                    (value) => {
                      const future =
                        Number(year) > today.getFullYear() ||
                        (Number(year) === today.getFullYear() &&
                          value >= today.getMonth() + 1);

                      return (
                        <option key={value} value={value} disabled={future}>
                          {getMonthName(value)}
                        </option>
                      );
                    },
                  )}
                </select>
              </div>

              <div className="sm:col-span-2 lg:col-span-4">
                <button
                  type="submit"
                  disabled={generating || !employeeId}
                  className="rounded-lg bg-blue-700 px-5 py-2.5 font-semibold text-white hover:bg-blue-800 disabled:cursor-not-allowed disabled:opacity-60"
                >
                  {generating ? "Generating..." : "Generate payroll"}
                </button>
              </div>
            </form>

            <p className="mt-4 text-xs leading-5 text-slate-500">
              The backend calculates salary, configured holiday exclusions,
              attendance, approved leave, loss of pay, and active salary
              components. Only draft payroll should be recalculated.
            </p>
          </section>
        )}

        <section className="grid gap-4 sm:grid-cols-2 lg:grid-cols-4">
          <SummaryCard
            label="Records shown"
            value={summary.count}
            icon={FileText}
          />
          <SummaryCard label="Draft" value={summary.draft} icon={Wallet} />
          <SummaryCard
            label="Locked"
            value={summary.locked}
            icon={LockKeyhole}
          />
          <SummaryCard
            label="Paid / simulated"
            value={summary.paid}
            icon={CircleDollarSign}
          />
        </section>

        <section className="rounded-2xl border border-slate-200 bg-white p-5 shadow-sm">
          <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-4">
            <div>
              <p className="text-sm text-slate-500">Net salary total</p>
              <p className="mt-1 text-xl font-bold">{money(summary.net)}</p>
              <p className="text-xs text-slate-500">
                Sum of records currently shown
              </p>
            </div>

            <div>
              <label
                htmlFor="payroll_search"
                className="mb-1.5 block text-sm font-semibold text-slate-700"
              >
                Search employee
              </label>
              <div className="flex items-center gap-2 rounded-lg border border-slate-300 px-3">
                <Search size={16} className="text-slate-400" />
                <input
                  id="payroll_search"
                  type="search"
                  value={search}
                  onChange={(event) => setSearch(event.target.value)}
                  placeholder="Name or employee code"
                  className="min-w-0 flex-1 py-2.5 text-sm outline-none"
                />
              </div>
            </div>

            <div>
              <label
                htmlFor="payroll_status"
                className="mb-1.5 block text-sm font-semibold text-slate-700"
              >
                Filter by status
              </label>
              <select
                id="payroll_status"
                value={statusFilter}
                onChange={(event) => setStatusFilter(event.target.value)}
                className="w-full rounded-lg border border-slate-300 bg-white px-3 py-2.5 text-sm"
              >
                <option value="ALL">All statuses</option>
                <option value="DRAFT">Draft</option>
                <option value="LOCKED">Locked</option>
                <option value="PAID">Paid</option>
              </select>
            </div>

            <div>
              <label
                htmlFor="payroll_period"
                className="mb-1.5 block text-sm font-semibold text-slate-700"
              >
                Filter by period
              </label>
              <select
                id="payroll_period"
                value={periodFilter}
                onChange={(event) => setPeriodFilter(event.target.value)}
                className="w-full rounded-lg border border-slate-300 bg-white px-3 py-2.5 text-sm"
              >
                <option value="ALL">All periods</option>
                {periods.map((period) => {
                  const [periodYear, periodMonth] = period.split("-");
                  return (
                    <option key={period} value={period}>
                      {getMonthName(periodMonth)} {periodYear}
                    </option>
                  );
                })}
              </select>
            </div>
          </div>
        </section>

        <section className="overflow-hidden rounded-2xl border border-slate-200 bg-white shadow-sm">
          <div className="flex flex-wrap items-center justify-between gap-3 border-b border-slate-200 p-5">
            <div>
              <h2 className="text-lg font-bold">
                {isAdminHR ? "Payroll records" : "My payroll history"}
              </h2>
              <p className="text-sm text-slate-500">
                {visibleRecords.length} record(s) shown
              </p>
            </div>

            <button
              type="button"
              onClick={refreshAll}
              disabled={loading}
              className="inline-flex items-center gap-2 rounded-lg border border-slate-300 px-3 py-2 text-sm font-semibold hover:bg-slate-50 disabled:opacity-50"
            >
              <RefreshCw size={16} className={loading ? "animate-spin" : ""} />
              Refresh
            </button>
          </div>

          {loading ? (
            <p className="p-8 text-center text-slate-500">
              Loading payroll records...
            </p>
          ) : visibleRecords.length === 0 ? (
            <p className="p-8 text-center text-slate-500">
              No payroll records match your filters.
            </p>
          ) : (
            <div className="overflow-x-auto">
              <table className="min-w-full divide-y divide-slate-200 text-sm">
                <thead className="bg-slate-50 text-left text-xs uppercase tracking-wide text-slate-500">
                  <tr>
                    {isAdminHR && (
                      <th className="px-4 py-3">Employee</th>
                    )}
                    <th className="px-4 py-3">Period</th>
                    <th className="px-4 py-3">Base salary</th>
                    <th className="px-4 py-3">Gross salary</th>
                    <th className="px-4 py-3">LOP days</th>
                    <th className="px-4 py-3">LOP deduction</th>
                    <th className="px-4 py-3">Total deductions</th>
                    <th className="px-4 py-3">Net salary</th>
                    <th className="px-4 py-3">Status</th>
                    <th className="px-4 py-3">Actions</th>
                  </tr>
                </thead>

                <tbody className="divide-y divide-slate-100">
                  {visibleRecords.map((record) => (
                    <tr
                      key={record.id}
                      className="align-middle hover:bg-slate-50"
                    >
                      {isAdminHR && (
                        <td className="px-4 py-4">
                          <p className="font-semibold text-slate-900">
                            {record.employee_name || "Employee"}
                          </p>
                          <p className="text-xs text-slate-500">
                            {record.employee_code || ""}
                          </p>
                        </td>
                      )}

                      <td className="whitespace-nowrap px-4 py-4 font-medium">
                        {getMonthName(record.month)} {record.year}
                      </td>
                      <td className="whitespace-nowrap px-4 py-4">
                        {money(record.base_salary)}
                      </td>
                      <td className="whitespace-nowrap px-4 py-4">
                        {money(record.gross_salary)}
                      </td>
                      <td className="px-4 py-4">{record.lop_days ?? 0}</td>
                      <td className="whitespace-nowrap px-4 py-4 text-red-700">
                        {money(record.lop_deduction)}
                      </td>
                      <td className="whitespace-nowrap px-4 py-4 text-red-700">
                        {money(record.total_deductions)}
                      </td>
                      <td className="whitespace-nowrap px-4 py-4 font-bold">
                        {money(record.net_salary)}
                      </td>
                      <td className="px-4 py-4">
                        <PayrollStatus status={record.status} />
                      </td>

                      <td className="min-w-64 px-4 py-4">
                        <div className="flex flex-wrap gap-2">
                          {isAdminHR && record.status === "DRAFT" && (
                            <button
                              type="button"
                              disabled={workingId !== null}
                              onClick={() =>
                                changePayrollStatus(record, "lock")
                              }
                              className="inline-flex items-center gap-1 rounded-lg bg-blue-700 px-3 py-2 text-xs font-bold text-white disabled:opacity-50"
                            >
                              <LockKeyhole size={14} />
                              {workingId === record.id
                                ? "Processing..."
                                : "Lock"}
                            </button>
                          )}

                          {isAdminHR && record.status === "LOCKED" && (
                            <button
                              type="button"
                              disabled={workingId !== null}
                              onClick={() =>
                                changePayrollStatus(record, "pay")
                              }
                              className="inline-flex items-center gap-1 rounded-lg bg-green-700 px-3 py-2 text-xs font-bold text-white disabled:opacity-50"
                            >
                              <CircleDollarSign size={14} />
                              {workingId === record.id
                                ? "Processing..."
                                : "Simulate pay"}
                            </button>
                          )}

                          {["LOCKED", "PAID"].includes(record.status) && (
                            <button
                              type="button"
                              disabled={downloadingId === record.id}
                              onClick={() => downloadSalarySlip(record)}
                              className="inline-flex items-center gap-1 rounded-lg border border-slate-300 px-3 py-2 text-xs font-bold text-slate-700 hover:bg-slate-50 disabled:opacity-50"
                            >
                              <Download size={14} />
                              {downloadingId === record.id
                                ? "Downloading..."
                                : "Salary slip"}
                            </button>
                          )}
                        </div>
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          )}
        </section>

        <p className="text-xs leading-5 text-slate-500">
          Simulated payments only record a payment event in this application.
          They do not transfer money through a bank or payment gateway.
          Payroll totals are based on the values returned by the backend.
        </p>
      </div>
    </main>
  );
}
